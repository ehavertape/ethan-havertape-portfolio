CIS 199 · PROGRAM 4

Student records: classes and encapsulation

A Student class that stores each student's data in private fields and only exposes it through properties. The properties enforce business rules: a GPA outside 0.0–4.0 or credit hours outside 0–12 are rejected and replaced with a default, so bad data can't get into the object. Enrollment status can only change through the Enroll() and Unenroll() methods.

  • Concepts: constructors, private backing fields, get/set properties with validation, methods, overriding ToString(), arrays of objects.

Student.cs

//Ethan Havertape
//CIS 199-01
//11-24-2024
//Program 4 - student data

using System;
using System.Collections.Generic;
using System.Linq;
using System.Text;
using System.Threading.Tasks;

namespace Program4
{
        public class Student //.cs file for student info
        {
            //create constant variables for gpa and credit hours
            public double DEFAULT_GPA = 4.0;
            public int DEFAULT_CREDITS = 12;

            //backing fields
            //all private variables for each
            private string _name;
            private double _gpa;
            private int _creditHours;
            private string _classification;
            private string _major;
            private bool enrollmentStatus; // private field for enrollment status

        //make constructor
        public Student(string name, double gpa, int creditHours, string classification, string major, bool enrollmentStatus)
            {
                Name = name;
                GPA = gpa;
                CreditHours = creditHours;
                Classification = classification;
                Major = major;
                this.enrollmentStatus = enrollmentStatus;
            }

            //properties
            public string Name //name properties get and set
            {
                get { return _name; }
                set { _name = value; }
            }

            public double GPA //gpa properties get and set
        {
                get { return _gpa; }
                set
                {
                    if (value >= 0.0 && value <= 4.0)
                        _gpa = value;
                    else
                        _gpa = DEFAULT_GPA;
                }
            }

            public int CreditHours //credit hours properties get and set
        {
                get { return _creditHours; }
                set
                {
                    if (value >= 0 && value <= 12)
                        _creditHours = value;
                    else
                        _creditHours = DEFAULT_CREDITS;
                }
            }

            public string Classification //classification properties get and set
        {
                get { return _classification; }
                set { _classification = value; }
            }


            public string Major //major properties get and set
        {
                get { return _major; }
                set { _major = value; }
            }

            //method to set enrollment status to true
            public void Enroll()
            { enrollmentStatus = true; }
            
            //method to set enrollment status to false
            public void Unenroll()
            { enrollmentStatus = false; }

            //method for currently enrolled
            public bool IsEnrolled()
            { return enrollmentStatus; }

            //display text with each variable on own line
            public override string ToString()
            {
                return $"Name: {Name}{Environment.NewLine}" +
                   $"GPA: {GPA:F1}{Environment.NewLine}" +
                   $"Credit Hours: {CreditHours}{Environment.NewLine}" +
                   $"Classification: {Classification}{Environment.NewLine}" +
                   $"Major: {Major}{Environment.NewLine}" +
                   $"Enrollment Status: {(enrollmentStatus ? "Enrolled" : "Not Enrolled")}{Environment.NewLine}";
            }
        }
    }

using System;

namespace Program4
{
    class Program //.cs file for program info (Main)
    {
        static void Main(string[] args)
        {
            // create an array of five Student objects
            Student[] students = new Student[]
            {
                new Student("John McLane", 3.6, 12, "Junior", "Computer Information Systems", true),
                new Student("Patrick Swayze", 3.8, 14, "Freshman", "Business Administration", true),
                new Student("Marty McFly", 3.3, 15, "Junior", "Chemistry", true),
                new Student("Dolly Parton", 3.6, 12, "Sophomore", "Mechanical Engineering", true),
                new Student("Harvey Spector", 2.8, 18, "Senior", "Criminal Justice", false)
            };

            // print current students data
            Console.WriteLine("--Current Students--");
            DisplayStudents(students);

            //// Make changes to properties and enrollment status
            students[0].Name = "John Marston"; //change name
            students[1].GPA = 3.7; //change gpa
            students[2].CreditHours = 9; //change credit hours
            students[3].Unenroll(); //change enrollment status 
            students[4].Enroll(); //change enrollment status

            // print after changes
            Console.WriteLine("--After Changes--");
            DisplayStudents(students);
        }

        // static method to print student data
        static void DisplayStudents(Student[] students)
        {
            foreach (Student student in students)
            {
                Console.WriteLine(student.ToString());
            }
        }
    }
}

CIS 199 · PROGRAM 3

Catering contract calculator: business logic in a GUI

A Windows Forms app that prices a catering contract. The user picks a caterer and a business and enters the contract length. The program looks up each caterer's discount rate and each business's base price from parallel arrays, then applies a tiered loyalty discount based on the number of years. It validates every input before calculating.

Try the logic

The original is a Windows desktop app. This is a small browser re-creation of the same pricing logic so you can test it here.

Form1.cs

//Ethan Havertape
//CIS 199-01
//Program 3  Catering Contract Calculator
//Due 11/7/2024

using System;
using System.Collections.Generic;
using System.ComponentModel;
using System.Data;
using System.Drawing;
using System.Linq;
using System.Text;
using System.Threading.Tasks;
using System.Windows.Forms;

namespace Program3
{
    public partial class Program3 : Form
    {
        //create arrays for the differect catering companies and their respective discounts
        string[] caterers =  { "Hill Catering Co.", "Food in a Flash", "Sally's Sandwiches", "Perry's Peirogis" };
        double[] discounts = { 0.30, 0.29, 0.12, 0.05 };

        //create arrays for different businesses and their respect contract prices
        string[] businesses = { "John's Books", "Office Supplies", "J.B. Car Parts", "Gevalia Coffee", "Ceylon Tea", "My Footwear" };
        double[] prices = { 500, 489, 412, 350, 325, 279 };

        //create arrays for the contract year ranges and their respective additional discounts/savings
        //use the ending year for the ranges
        int[] yearRanges = { 1, 4, 7, 8 };
        double[] savings = { 0, 30, 40, 50 };


        public Program3()
        {
            InitializeComponent();
        }

        private void CalcBtn_Click(object sender, EventArgs e)
        {
            //create a message box if the user does not select a caterer
            if (catererComBox.SelectedIndex < 0)
            {
                MessageBox.Show("Please select Caterer");
                return;
            }
            
            //create a message box if the user does not select a business
            if (businessComBox.SelectedIndex < 0)
            {
                MessageBox.Show("Please select Business");
                return;
            }

            //create a message box if the user does not enter a contract year, or an invalid year/input
            //use tryparse for this
            if (!int.TryParse(yearsTxt.Text, out int contractYears) || (contractYears < 0))
            {
                MessageBox.Show("Please provide valid contract years");
                return;
            }

            //code for finding the discount rate for the caterer selected
            int catererIndex = catererComBox.SelectedIndex;
            double discountRate = discounts[catererIndex];

            //find the contract price for the business selected
            int businessIndex = businessComBox.SelectedIndex;
            double contractPrice = prices[businessIndex];

            //caclulate the discount amount by multiplying the contract price and the discount rate
            double discountAmt = contractPrice * discountRate;

            //find the additional savings/discounts through the contract years
            //use for loop
            //else if to calculate for years 8+
            double additionalDiscount = 0;
            for (int i = 0; i < yearRanges.Length; i++)
            {
                if (contractYears < yearRanges[i])
                {
                    additionalDiscount = savings[i];
                    break;
                }

                else if (contractYears >= 8)
                {
                    additionalDiscount = savings[savings.Length - 1];
                    break;
                }

            }

            //calcluate total price by factoring in additional discounts/savings
            double totalPrice = contractPrice - discountAmt - additionalDiscount;

            //display price
            priceTxt.Text = $"${totalPrice:F2}";
        }
    }
}
